// ─────────────────────────────────────────────────────────────────────────────
// Процедура для расширения ВыгрузкаВсехВнешнихОбработок (СЕРВЕРНАЯ часть).
// ─────────────────────────────────────────────────────────────────────────────
//
// Добавить в общий модуль:
//   Расш2_ВыгрузкаДопОбработокИОтчетовСервер
//
// Функция возвращает Структуру со структурой хранения справочника
// ДополнительныеОтчетыИОбработки. Запись в JSON делается на клиенте
// (см. ManagedApplicationModule_patch.bsl.txt) — сервер 1С может физически
// находиться на другой машине и не иметь доступа к каталогам клиента.
//
// ─────────────────────────────────────────────────────────────────────────────

&НаСервере
Функция ПолучитьСтруктуруХраненияСправочникаДопОбработок() Экспорт

	// ПолучитьСтруктуруХраненияБазыДанных:
	//   Параметр 1 — МАССИВ объектов метаданных.
	//   Параметр 2 — Булево (ВключатьСлужебныеТаблицы), по умолчанию Ложь.
	МассивОбъектов = Новый Массив;
	МассивОбъектов.Добавить(Метаданные.Справочники.ДополнительныеОтчетыИОбработки);
	ТЗ = ПолучитьСтруктуруХраненияБазыДанных(МассивОбъектов, Ложь);

	// Ищем основную таблицу (не «Изменения», не табличные части).
	ОсновнаяТаблица = Неопределено;
	Для Каждого СтрокаТаблицы Из ТЗ Цикл
		Если СтрокаТаблицы.Назначение = "Основная" Тогда
			ОсновнаяТаблица = СтрокаТаблицы;
			Прервать;
		КонецЕсли;
	КонецЦикла;

	Если ОсновнаяТаблица = Неопределено Тогда
		ВызватьИсключение "Не найдена основная таблица справочника ДополнительныеОтчетыИОбработки";
	КонецЕсли;

	// Массив всех физических колонок. Одному реквизиту 1С может соответствовать
	// несколько физических полей — например, ссылочное поле "Вид" разворачивается
	// в _FldNNNRRef (binary UUID), составной — в _FldNNN_TYPE / _FldNNN_RTRef /
	// _FldNNN_RRRef / _FldNNN_N / _FldNNN_S и т.п. Клиент (Rust) сам выбирает нужную.
	Поля = Новый Массив;
	Для Каждого Поле Из ОсновнаяТаблица.Поля Цикл
		Запись = Новый Структура;
		Запись.Вставить("name", Строка(Поле.ИмяПоля));
		Запись.Вставить("storage", Строка(Поле.ИмяПоляХранения));
		Поля.Добавить(Запись);
	КонецЦикла;

	Результат = Новый Структура;
	Результат.Вставить("table", ОсновнаяТаблица.ИмяТаблицыХранения);
	Результат.Вставить("fields", Поля);
	Возврат Результат;

КонецФункции
